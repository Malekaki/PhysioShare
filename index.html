Step 1: Open script.google.com & Paste This Controller Script (Code.gs)

  1. Go to https://script.google.com while signed into your Google One (2 TB) account and click New project.
  2. Delete any placeholder code in Code.gs, paste the script below, and click the Save icon:
/**
 * PHYSIOSHARE INVISIBLE CLOUD STORAGE CONTROLLER (Code.gs)
 * Supports Direct Resumable Binary Streaming (createUploadSession)
 * + Chunked Fallback so large files never hit argument size limits.
 */
var ROOT_FOLDER_NAME = "PhysioShare_Library";
var MODULE_FOLDERS = [
  "Foundations of Anatomy and Clinical Practice",
  "Cardiorespiratory",
  "Musculoskeletal"
];

function doGet(e) {
  try {
    var params = (e && e.parameter) ? e.parameter : {};
    var action = params.action || "ping";

    if (action === "ping") {
      var info = ensureModuleFoldersExist_();
      return jsonOutput({
        ok: true,
        success: true,
        status: "connected",
        rootFolderName: ROOT_FOLDER_NAME,
        folders: info.folders
      });
    }

    if (action === "list") {
      var items = listAllModuleFiles_(params.module);
      return jsonOutput({ ok: true, success: true, files: items });
    }

    if (action === "download" && params.fileId) {
      return handleDownload(params.fileId);
    }

    return jsonOutput({ ok: true, success: true, message: "PhysioShare Cloud Storage is online." });
  } catch (err) {
    return jsonOutput({ ok: false, success: false, error: err.toString() });
  }
}

function doPost(e) {
  try {
    var rawBody = (e && e.postData && e.postData.contents) ? e.postData.contents : "{}";
    var data = JSON.parse(rawBody);
    var action = data.action || "createUploadSession";

    if (action === "ping") {
      var folderInfo = ensureModuleFoldersExist_();
      return jsonOutput({
        ok: true,
        success: true,
        rootFolderName: ROOT_FOLDER_NAME,
        folders: folderInfo.folders
      });
    }

    if (action === "list") {
      return jsonOutput({ ok: true, success: true, files: listAllModuleFiles_(data.module) });
    }

    if (action === "download" && data.fileId) {
      return handleDownload(data.fileId);
    }

    // 1. Direct Resumable Upload Session (Bypasses Apps Script argument size limits)
    if (action === "createUploadSession") {
      return handleCreateUploadSession(data);
    }

    // 2. Chunked Upload Support (Automatic Fallback)
    if (action === "uploadChunk") {
      return handleChunkedUpload_(data);
    }

    // 3. Small File Direct Upload
    if (action === "upload" || action === "uploadFile") {
      return handleSmallUpload_(data);
    }

    return jsonOutput({ ok: false, success: false, error: "Unsupported POST action: " + action });
  } catch (err) {
    return jsonOutput({ ok: false, success: false, error: err.toString() });
  }
}

// 1. Generate Direct Upload Session (Bypasses Apps Script parameter limits)
function handleCreateUploadSession(data) {
  var modName = data.moduleName || data.module || "Foundations of Anatomy and Clinical Practice";
  var targetFolder = getOrCreateModuleFolder(modName);
  var targetFolderId = targetFolder.getId();

  var url = "https://www.googleapis.com/upload/drive/v3/files?uploadType=resumable";
  var descMeta = JSON.stringify({
    title: data.title || data.fileName,
    description: data.description || "",
    authorName: data.authorName || "Anonymous Student",
    moduleName: modName,
    resourceType: data.resourceType || "document"
  });

  var metadata = {
    name: data.fileName,
    parents: [targetFolderId],
    description: descMeta
  };

  var headers = {
    "Authorization": "Bearer " + ScriptApp.getOAuthToken(),
    "Content-Type": "application/json; charset=UTF-8"
  };
  if (data.origin && data.origin !== "null") {
    headers["Origin"] = data.origin;
  }

  var options = {
    method: "POST",
    headers: headers,
    payload: JSON.stringify(metadata),
    muteHttpExceptions: true
  };

  var response = UrlFetchApp.fetch(url, options);
  var respHeaders = response.getAllHeaders();
  var locationUrl = respHeaders["Location"] || respHeaders["location"];

  if (locationUrl) {
    return jsonOutput({ ok: true, success: true, uploadUrl: locationUrl });
  } else {
    return jsonOutput({ ok: false, success: false, error: "Drive session failed: " + response.getContentText() });
  }
}

// 2. Chunked Upload Fallback (Writes slices to temp Drive file, never exceeding 2MB limit)
function handleChunkedUpload_(data) {
  var modName = data.moduleName || data.module || "Foundations of Anatomy and Clinical Practice";
  var targetFolder = getOrCreateModuleFolder(modName);
  var props = PropertiesService.getScriptProperties();
  var tempKey = "TEMP_UPLOAD_" + data.uploadId;

  if (data.chunkIndex === 0) {
    var firstBytes = Utilities.base64Decode(data.chunkData);
    var firstBlob = Utilities.newBlob(firstBytes, data.mimeType || "application/octet-stream", data.fileName);
    var created = targetFolder.createFile(firstBlob);
    props.setProperty(tempKey, created.getId());
    if (!data.isLastChunk) {
      return jsonOutput({ ok: true, success: true, chunkStored: 0 });
    }
  }

  var fileId = props.getProperty(tempKey);
  if (!fileId) {
    return jsonOutput({ ok: false, success: false, error: "Upload session expired." });
  }

  var existingFile = DriveApp.getFileById(fileId);
  if (data.chunkIndex > 0) {
    var existingBytes = existingFile.getBlob().getBytes();
    var nextBytes = Utilities.base64Decode(data.chunkData);
    var combined = existingBytes.concat(nextBytes);
    var newBlob = Utilities.newBlob(combined, data.mimeType || "application/octet-stream", data.fileName);
    var updatedFile = targetFolder.createFile(newBlob);
    existingFile.setTrashed(true);
    fileId = updatedFile.getId();
    props.setProperty(tempKey, fileId);
    existingFile = updatedFile;
  }

  if (data.isLastChunk) {
    props.deleteProperty(tempKey);
    var descMeta = JSON.stringify({
      title: data.title || data.fileName,
      description: data.description || "",
      authorName: data.authorName || "Anonymous Student",
      moduleName: modName,
      resourceType: data.resourceType || "document"
    });
    existingFile.setDescription(descMeta);
    existingFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return jsonOutput({
      ok: true,
      success: true,
      fileId: existingFile.getId(),
      name: existingFile.getName(),
      fileName: existingFile.getName()
    });
  }

  return jsonOutput({ ok: true, success: true, chunkStored: data.chunkIndex });
}

function handleSmallUpload_(data) {
  var modName = data.moduleName || data.module || "Foundations of Anatomy and Clinical Practice";
  var targetFolder = getOrCreateModuleFolder(modName);
  var rawB64 = data.base64Data || data.fileData || "";
  var bytes = Utilities.base64Decode(rawB64);
  var blob = Utilities.newBlob(bytes, data.mimeType || "application/octet-stream", data.fileName);
  var createdFile = targetFolder.createFile(blob);
  var descMeta = JSON.stringify({
    title: data.title || data.fileName,
    description: data.description || "",
    authorName: data.authorName || "Anonymous Student",
    moduleName: modName,
    resourceType: data.resourceType || "document"
  });
  createdFile.setDescription(descMeta);
  createdFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return jsonOutput({
    ok: true,
    success: true,
    fileId: createdFile.getId(),
    name: createdFile.getName(),
    fileName: createdFile.getName()
  });
}

function getOrCreateModuleFolder(moduleName) {
  var rootFolders = DriveApp.getFoldersByName(ROOT_FOLDER_NAME);
  var root = rootFolders.hasNext() ? rootFolders.next() : DriveApp.createFolder(ROOT_FOLDER_NAME);
  var modFolders = root.getFoldersByName(moduleName);
  return modFolders.hasNext() ? modFolders.next() : root.createFolder(moduleName);
}

function ensureModuleFoldersExist_() {
  var result = {};
  for (var i = 0; i < MODULE_FOLDERS.length; i++) {
    var modName = MODULE_FOLDERS[i];
    var sub = getOrCreateModuleFolder(modName);
    result[modName] = { id: sub.getId() };
  }
  return { folders: result };
}

function listAllModuleFiles_(requestedModule) {
  var out = [];
  var targets = requestedModule ? [requestedModule] : MODULE_FOLDERS;
  for (var i = 0; i < targets.length; i++) {
    var modName = targets[i];
    var sub = getOrCreateModuleFolder(modName);
    var files = sub.getFiles();
    while (files.hasNext()) {
      var f = files.next();
      if (!f.isTrashed()) {
        var fid = f.getId();
        var rawDesc = f.getDescription() || "";
        var meta = {};
        try { meta = JSON.parse(rawDesc); } catch (e) { meta = { title: rawDesc }; }
        out.push({
          id: fid,
          fileId: fid,
          name: f.getName(),
          fileName: f.getName(),
          size: f.getSize(),
          fileSize: Math.max(1, Math.round(f.getSize() / 1024)) + " KB",
          mimeType: f.getMimeType(),
          moduleName: modName,
          title: meta.title || f.getName().replace(/\.[^/.]+$/, ""),
          description: meta.description || "",
          authorName: meta.authorName || "PhysioShare Library",
          resourceType: meta.resourceType || "",
          createdTime: f.getDateCreated().toISOString(),
          dateCreated: f.getDateCreated().toISOString()
        });
      }
    }
  }
  return out;
}

function handleDownload(fileId) {
  try {
    var file = DriveApp.getFileById(fileId);
    var blob = file.getBlob();
    var b64 = Utilities.base64Encode(blob.getBytes());
    return jsonOutput({
      ok: true,
      success: true,
      name: file.getName(),
      fileName: file.getName(),
      mimeType: blob.getContentType(),
      base64: b64,
      base64Data: b64
    });
  } catch (err) {
    return jsonOutput({ ok: false, success: false, error: "Download failed: " + err.toString() });
  }
}

function jsonOutput(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
