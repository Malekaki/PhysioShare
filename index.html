Google Drive (2 TB) Setup & Anonymous Student Upload Bridge

Allow any student with the website passcode to upload & download files directly from your 2 TB Google Drive at £0 cost

Google Drive 2 TB Bridge Status Paste your Google Apps Script Web App URL in Step 4 below to enable direct 2 TB Google Drive storage for all students.
STEP 1

How Anonymous Student Uploads to Your 2 TB Google Drive Work

Normally, Google Drive requires someone to sign into a Google account to upload files. To let your classmates upload without logging into Google (and without ever seeing your Google password or private Drive files), we use a free Google Apps Script Web App attached to your Google account:

  • Runs as "Me" (Your Google Account): The script has permission to save uploaded files directly into your 2 TB Google Drive inside PhysioShare_2TB_Library/{Module Name}/.
  • Accessible by "Anyone": When a student selects a file on PhysioShare and clicks Publish Resource, the website sends the file to your Web App URL via fetch() POST.
  • Instant Public Preview & Download: The script saves the file into your Drive, sets that file's permission to Anyone with the link can view/download, and returns the Google Drive fileId, viewUrl, and downloadUrl to PhysioShare so every student on any computer can stream audio, preview PDFs, and download it immediately.
STEP 2

Create Your Google Apps Script at script.google.com

  1. Open https://script.google.com while signed into the Google account that has your 2 TB Google One storage.
  2. Click New project (top-left button) and rename the project at the top to PhysioShare Drive Bridge.
  3. Delete the default function myFunction() {} code inside the editor, click Copy Complete Code.gs Script above, and paste the entire script in.
  4. Click the Save project icon (floppy disk icon or press Ctrl+S / Cmd+S).
/**
 * PHYSIOSHARE 2 TB GOOGLE DRIVE BRIDGE (Code.gs)
 * Allows any student with the PhysioShare passcode to upload, preview,
 * and download resources directly from your 2 TB Google Drive at £0 cost.
 */
var ROOT_FOLDER_NAME = "PhysioShare_2TB_Library";
var MODULE_FOLDERS = [
  "Foundations of Anatomy and Clinical Practice",
  "Cardiorespiratory",
  "Musculoskeletal"
];

function doGet(e) {
  try {
    var params = (e && e.parameter) ? e.parameter : {};
    var action = params.action || "ping";

    if (action === "ping") {
      var info = ensureModuleFoldersExist_();
      return jsonResponse_({
        ok: true,
        status: "connected",
        rootFolderName: ROOT_FOLDER_NAME,
        rootFolderUrl: info.rootUrl,
        folders: info.folders
      });
    }

    if (action === "list") {
      var items = listAllModuleFiles_();
      return jsonResponse_({ ok: true, files: items });
    }

    return jsonResponse_({ ok: true, message: "PhysioShare Google Drive 2 TB Bridge is online." });
  } catch (err) {
    return jsonResponse_({ ok: false, error: String(err) });
  }
}

function doPost(e) {
  try {
    var props = PropertiesService.getScriptProperties();
    var rawBody = (e && e.postData && e.postData.contents) ? e.postData.contents : "{}";
    var data = JSON.parse(rawBody);
    var action = data.action || "upload";

    // 1. Optional Backend Passcode Verification
    if (action === "verifyPasscode") {
      var storedHash = props.getProperty("SHARED_PASSCODE_HASH") || "";
      var isValid = storedHash ? (data.passcodeHash === storedHash) : true;
      return jsonResponse_({ ok: isValid });
    }

    // 2. Optional Backend Administrator Verification
    if (action === "verifyAdmin") {
      var adminHash = props.getProperty("ADMIN_PASSCODE_HASH") || "";
      var isAdmin = adminHash ? (data.passcodeHash === adminHash) : false;
      return jsonResponse_({ ok: isAdmin });
    }

    // 3. Ping / Create Module Folders in Owner's 2 TB Google Drive
    if (action === "ping") {
      var folderInfo = ensureModuleFoldersExist_();
      return jsonResponse_({
        ok: true,
        rootFolderName: ROOT_FOLDER_NAME,
        rootFolderUrl: folderInfo.rootUrl,
        folders: folderInfo.folders
      });
    }

    // 4. List Existing Files in Owner's Google Drive Module Folders
    if (action === "list") {
      return jsonResponse_({ ok: true, files: listAllModuleFiles_() });
    }

    // 5. Anonymous Student Upload Directly into Owner's 2 TB Google Drive
    if (action === "upload" || action === "uploadFile") {
      var rootFolder = getOrCreateFolder_(ROOT_FOLDER_NAME);
      var targetModule = data.moduleName || "Foundations of Anatomy and Clinical Practice";
      var moduleFolder = getOrCreateSubFolder_(rootFolder, targetModule);

      var bytes = Utilities.base64Decode(data.base64Data);
      var mimeType = data.mimeType || "application/octet-stream";
      var fileName = data.fileName || ("PhysioShare_Resource_" + new Date().getTime());
      var blob = Utilities.newBlob(bytes, mimeType, fileName);

      var driveFile = moduleFolder.createFile(blob);
      var descMeta = JSON.stringify({
        title: data.title || fileName,
        description: data.description || "",
        authorName: data.authorName || "Anonymous Student",
        moduleName: targetModule,
        resourceType: data.resourceType || "document"
      });
      driveFile.setDescription(descMeta);

      // Make file viewable/downloadable by anyone with the link so all students can preview/download
      driveFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

      var fileId = driveFile.getId();
      return jsonResponse_({
        ok: true,
        fileId: fileId,
        fileName: driveFile.getName(),
        viewUrl: "https://drive.google.com/file/d/" + fileId + "/preview",
        openUrl: "https://drive.google.com/file/d/" + fileId + "/view",
        downloadUrl: "https://drive.google.com/uc?export=download&id=" + fileId,
        audioStreamUrl: "https://drive.google.com/uc?export=download&id=" + fileId,
        moduleFolderUrl: moduleFolder.getUrl()
      });
    }

    return jsonResponse_({ ok: false, error: "Unrecognised action: " + action });
  } catch (err) {
    return jsonResponse_({ ok: false, error: String(err) });
  }
}

function ensureModuleFoldersExist_() {
  var root = getOrCreateFolder_(ROOT_FOLDER_NAME);
  root.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  var result = {};
  for (var i = 0; i < MODULE_FOLDERS.length; i++) {
    var modName = MODULE_FOLDERS[i];
    var sub = getOrCreateSubFolder_(root, modName);
    sub.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    result[modName] = { id: sub.getId(), url: sub.getUrl() };
  }
  return { rootId: root.getId(), rootUrl: root.getUrl(), folders: result };
}

function listAllModuleFiles_() {
  var root = getOrCreateFolder_(ROOT_FOLDER_NAME);
  var out = [];
  for (var i = 0; i < MODULE_FOLDERS.length; i++) {
    var modName = MODULE_FOLDERS[i];
    var sub = getOrCreateSubFolder_(root, modName);
    var files = sub.getFiles();
    while (files.hasNext()) {
      var f = files.next();
      var fid = f.getId();
      var meta = {};
      try { meta = JSON.parse(f.getDescription() || "{}"); } catch (e) {}
      out.push({
        fileId: fid,
        fileName: f.getName(),
        fileSize: Math.max(1, Math.round(f.getSize() / 1024)) + " KB",
        mimeType: f.getMimeType(),
        moduleName: modName,
        title: meta.title || f.getName().replace(/\.[^/.]+$/, ""),
        description: meta.description || "",
        authorName: meta.authorName || "Google Drive Library",
        resourceType: meta.resourceType || "",
        dateCreated: f.getDateCreated().toISOString(),
        viewUrl: "https://drive.google.com/file/d/" + fid + "/preview",
        downloadUrl: "https://drive.google.com/uc?export=download&id=" + fid
      });
    }
  }
  return out;
}

function jsonResponse_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateFolder_(name) {
  var folders = DriveApp.getFoldersByName(name);
  return folders.hasNext() ? folders.next() : DriveApp.createFolder(name);
}

function getOrCreateSubFolder_(parent, name) {
  var folders = parent.getFoldersByName(name);
  return folders.hasNext() ? folders.next() : parent.createFolder(name);
}
STEP 3

Crucial Deployment Settings (So Other Students Can Upload)

In the Google Apps Script editor, follow these exact clicks so students do not need a Google account or permission prompt when uploading:

  1. Click the blue Deploy button (top-right corner) → select New deployment.
  2. Click the gear icon next to Select type → choose Web app.
  3. Set the two dropdowns exactly as follows:
    Execute as Me (your Google email) Ensures all student uploads use your 2 TB Google One storage quota.
    Who has access Anyone Essential so anonymous students on any computer can upload without signing into Google.
  4. Click Deploy → click Authorize access → select your Google account.
  5. If Google shows "Google hasn't verified this app", click Advanced (bottom-left) → click Go to PhysioShare Drive Bridge (unsafe) → click Allow. (This warning appears simply because you wrote the script yourself in your own account.)
  6. Copy the generated Web app URL (it starts with https://script.google.com/macros/s/.../exec).

STEP 5

Automatic Google Drive Folder Structure & Managing Files

Your Google Apps Script automatically organises all student uploads inside your Google Drive using this clean hierarchy:

📁 PhysioShare_2TB_Library (Root folder in your My Drive)
├── 📁 Foundations of Anatomy and Clinical Practice
├── 📁 Cardiorespiratory
└── 📁 Musculoskeletal
  • Adding Files Directly in Google Drive: You can also drag and drop PDFs, audio files, or lecture slides directly into any of those 3 folders inside your Google Drive app, then click Sync Existing Drive Files on PhysioShare to pull them straight into the website list.
  • Read-Only Protection for Students: Students can upload new files, stream audio, preview PDFs, and download files, but the script never grants students permission to delete or overwrite existing files in your Google Drive.
