/**
 * PHYSIOSHARE INVISIBLE CLOUD STORAGE CONTROLLER (Code.gs)
 * Compatible with GitHub Pages static hosting via CORS-safe text/plain POST payloads.
 * Supports single-shot uploads, chunked/resumable uploads, and direct base64 file streaming
 * so users never leave PhysioShare or see Google Drive links.
 */
var ROOT_FOLDER_NAME = "PhysioShare_2TB_Library";
var MODULE_FOLDERS = [
  "Foundations of Anatomy and Clinical Practice",
  "Cardiorespiratory",
  "Musculoskeletal"
];

function doGet(e) {
  try {
    var params = (e && e.parameter) ? e.parameter : {};
    var action = params.action || "ping";

    if (action === "ping") {
      var info = ensureModuleFoldersExist_();
      return jsonResponse_({
        ok: true,
        status: "connected",
        folderCount: info.count
      });
    }

    if (action === "list") {
      return jsonResponse_({ ok: true, files: listAllModuleFiles_() });
    }

    if (action === "download" && params.fileId) {
      return streamFileAsBase64_(params.fileId);
    }

    return jsonResponse_({ ok: true, message: "PhysioShare Cloud Storage Backend Online" });
  } catch (err) {
    return jsonResponse_({ ok: false, error: String(err) });
  }
}

function doPost(e) {
  try {
    var rawBody = (e && e.postData && e.postData.contents) ? e.postData.contents : "{}";
    var data = JSON.parse(rawBody);
    var action = data.action || "upload";

    if (action === "ping") {
      var info = ensureModuleFoldersExist_();
      return jsonResponse_({ ok: true, status: "connected", folderCount: info.count });
    }

    if (action === "list") {
      return jsonResponse_({ ok: true, files: listAllModuleFiles_() });
    }

    if (action === "download" && data.fileId) {
      return streamFileAsBase64_(data.fileId);
    }

    // Standard single-payload upload
    if (action === "upload") {
      var rootFolder = getOrCreateFolder_(ROOT_FOLDER_NAME);
      var targetModule = data.moduleName || MODULE_FOLDERS[0];
      var moduleFolder = getOrCreateSubFolder_(rootFolder, targetModule);

      var bytes = Utilities.base64Decode(data.base64Data);
      var mimeType = data.mimeType || "application/octet-stream";
      var fileName = data.fileName || ("PhysioShare_Resource_" + new Date().getTime());
      var blob = Utilities.newBlob(bytes, mimeType, fileName);

      var driveFile = moduleFolder.createFile(blob);
      var descMeta = JSON.stringify({
        title: data.title || fileName,
        description: data.description || "",
        authorName: data.authorName || "Anonymous Student",
        moduleName: targetModule,
        resourceType: data.resourceType || "document",
        uploadedAt: new Date().toISOString()
      });
      driveFile.setDescription(descMeta);

      return jsonResponse_({
        ok: true,
        fileId: driveFile.getId(),
        fileName: driveFile.getName(),
        fileSize: driveFile.getSize(),
        uploadedAt: driveFile.getDateCreated().toISOString()
      });
    }

    // Chunked / Resumable Upload for larger files
    if (action === "uploadChunk") {
      var cache = CacheService.getScriptCache();
      var uploadId = data.uploadId;
      var chunkIndex = Number(data.chunkIndex || 0);
      var totalChunks = Number(data.totalChunks || 1);
      cache.put("chunk_" + uploadId + "_" + chunkIndex, data.chunkBase64, 21600);

      if (chunkIndex + 1 < totalChunks) {
        return jsonResponse_({ ok: true, chunkReceived: chunkIndex });
      }

      // Reassemble all chunks on final chunk
      var combinedBase64 = "";
      for (var c = 0; c < totalChunks; c++) {
        var part = cache.get("chunk_" + uploadId + "_" + c);
        if (!part) {
          return jsonResponse_({ ok: false, error: "Missing upload chunk " + c });
        }
        combinedBase64 += part;
        cache.remove("chunk_" + uploadId + "_" + c);
      }

      var root = getOrCreateFolder_(ROOT_FOLDER_NAME);
      var modName = data.moduleName || MODULE_FOLDERS[0];
      var modFolder = getOrCreateSubFolder_(root, modName);

      var decodedBytes = Utilities.base64Decode(combinedBase64);
      var fileBlob = Utilities.newBlob(decodedBytes, data.mimeType || "application/octet-stream", data.fileName);
      var created = modFolder.createFile(fileBlob);

      created.setDescription(JSON.stringify({
        title: data.title || data.fileName,
        description: data.description || "",
        authorName: data.authorName || "Anonymous Student",
        moduleName: modName,
        resourceType: data.resourceType || "document",
        uploadedAt: new Date().toISOString()
      }));

      return jsonResponse_({
        ok: true,
        fileId: created.getId(),
        fileName: created.getName(),
        fileSize: created.getSize(),
        uploadedAt: created.getDateCreated().toISOString()
      });
    }

    return jsonResponse_({ ok: false, error: "Unsupported action: " + action });
  } catch (err) {
    return jsonResponse_({ ok: false, error: String(err) });
  }
}

function streamFileAsBase64_(fileId) {
  var file = DriveApp.getFileById(fileId);
  var blob = file.getBlob();
  return jsonResponse_({
    ok: true,
    fileId: file.getId(),
    fileName: file.getName(),
    mimeType: blob.getContentType() || "application/octet-stream",
    base64Data: Utilities.base64Encode(blob.getBytes())
  });
}

function ensureModuleFoldersExist_() {
  var root = getOrCreateFolder_(ROOT_FOLDER_NAME);
  for (var i = 0; i < MODULE_FOLDERS.length; i++) {
    getOrCreateSubFolder_(root, MODULE_FOLDERS[i]);
  }
  return { count: MODULE_FOLDERS.length };
}

function listAllModuleFiles_() {
  var root = getOrCreateFolder_(ROOT_FOLDER_NAME);
  var out = [];
  for (var i = 0; i < MODULE_FOLDERS.length; i++) {
    var modName = MODULE_FOLDERS[i];
    var sub = getOrCreateSubFolder_(root, modName);
    var files = sub.getFiles();
    while (files.hasNext()) {
      var f = files.next();
      if (f.isTrashed()) continue;
      var meta = {};
      try { meta = JSON.parse(f.getDescription() || "{}"); } catch (e) {}
      out.push({
        fileId: f.getId(),
        fileName: f.getName(),
        fileSize: Math.max(1, Math.round(f.getSize() / 1024)) + " KB",
        rawBytes: f.getSize(),
        mimeType: f.getMimeType(),
        moduleName: modName,
        title: meta.title || f.getName().replace(/\.[^/.]+$/, ""),
        description: meta.description || "",
        authorName: meta.authorName || "PhysioShare Library",
        resourceType: meta.resourceType || "",
        dateCreated: f.getDateCreated().toISOString()
      });
    }
  }
  return out;
}

function jsonResponse_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateFolder_(name) {
  var folders = DriveApp.getFoldersByName(name);
  return folders.hasNext() ? folders.next() : DriveApp.createFolder(name);
}

function getOrCreateSubFolder_(parent, name) {
  var folders = parent.getFoldersByName(name);
  return folders.hasNext() ? folders.next() : parent.createFolder(name);
}
